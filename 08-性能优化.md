## 8.1 JVM 调优

### 8.1.1 常用 JVM 参数

```bash

-XX:MetaspaceSize=256m
-XX:MaxMetaspaceSize=512m

# GC 选择
-XX:+UseG1GC
-XX:+UseZGC

# GC 日志
-XX:+PrintGCDetails
-XX:+PrintGCDateStamps
-Xlog:gc*             # JDK 9+

# OOM 处理
-XX:+HeapDumpOnOutOfMemoryError
-XX:HeapDumpPath=/logs/dump.hprof

# 调试
-agentlib:jdwp=transport=dt_socket,address=5005,server=y,suspend=n
```

## 8.2 代码层面优化

### 8.2.1 对象创建优化

```java
public class ObjectOptimization {
    // 1. 重用对象，避免频繁创建
    public void reuseObject() {
        StringBuilder sb = new StringBuilder(); // 可复用
        for (int i = 0; i < 10000; i++) {
            sb.setLength(0); // 清空重用
            sb.append(i);
        }
    }
    
    // 2. 使用对象池
    private static final ObjectPool<Connection> pool = new ObjectPool<>(
        Connection::create, Connection::close, 10
    );
    
    // 3. 基本类型优先于包装类型
    int sum = 0;           // 基本类型
    Integer boxedSum = 0;  // 包装类型，有装箱拆箱开销
    
    // 4. 避免不必要的自动装箱拆箱
    public long calculateSum(List<Integer> list) {
        long sum = 0L;
        for (Integer val : list) {
            sum += val; // 自动拆箱
        }
        return sum;
    }
    
    // 5. 使用享元模式复用对象
    public class FlyweightFactory {
        private final Map<String, Flyweight> pool = new ConcurrentHashMap<>();
        
        public Flyweight getFlyweight(String key) {
            return pool.computeIfAbsent(key, k -> new Flyweight(k));
        }
    }
}
```

### 8.2.2 集合使用优化

```java
public class CollectionOptimization {
    // 1. 指定集合初始容量
    List<String> list = new ArrayList<>(10000);
    Map<String, Object> map = new HashMap<>(16);
    
    // 2. 选择合适的集合
    // 随机访问：ArrayList > LinkedList
    // 频繁插入删除：LinkedList > ArrayList
    // 去重：HashSet/LinkedHashSet/TreeSet
    // 排序：TreeSet/Stream.sorted()
    
    // 3. 使用并发集合
    Map<String, Integer> concurrentMap = new ConcurrentHashMap<>();
    
    // 4. 使用原始类型集合（FastUtil/HPPC）
    // it.unimi.dsi.fastutil
    IntArrayList intList = new IntArrayList();
    intList.add(42); // 避免装箱
    
    // 5. 遍历优化
    public void iterateOptimization(List<String> list) {
        // 推荐：for-each（编译器优化为iterator）
        for (String item : list) { /* ... */ }
        
        // 或：使用索引（对于ArrayList）
        for (int i = 0, size = list.size(); i < size; i++) {
            String item = list.get(i);
        }
        
        // 避免：iterator.next()在循环内重复调用
    }
}
```

### 8.2.3 字符串优化

```java
public class StringOptimization {
    // 1. 字符串拼接使用 StringBuilder
    public String concatStrings(List<String> strings) {
        StringBuilder sb = new StringBuilder(strings.size() * 20);
        for (String s : strings) {
            sb.append(s);
        }
        return sb.toString();
    }
    
    // 2. 字符串常量池
    String s1 = "hello";           // 常量池
    String s2 = "hello";           // 同一引用
    String s3 = new String("hello"); // 新对象
    
    System.out.println(s1 == s2);  // true
    System.out.println(s1 == s3);  // false
    
    // 3. intern() 方法
    String s4 = new String("world").intern(); // 放入常量池
    
    // 4. 字符串格式化
    // 简单拼接：使用 + 或 StringBuilder
    // 复杂格式化：使用 String.format()，但开销较大
    String formatted = String.format("Name: %s, Age: %d", name, age);
    
    // 5. 正则表达式预编译
    private static final Pattern PATTERN = Pattern.compile("\\d+");
    
    public boolean matches(String input) {
        return PATTERN.matcher(input).matches();
    }
}
```

### 8.2.4 其他优化技巧

```java
public class GeneralOptimization {
    // 1. 避免方法在循环中被反射调用
    // 缓存 Method/Constructor/Field 引用
    
    // 2. 使用局部变量
    public void useLocalVariables() {
        long start = System.nanoTime();
        int sum = 0;
        for (int i = 0; i < 1_000_000; i++) {
            sum += i; // 局部变量，直接从栈访问
        }
    }
    
    // 3. 注意方法内联
    // JIT 会自动内联小方法
    // 可使用 -XX:+PrintInlining 查看
    
    // 4. 避免创建不必要的异常
    // 异常构造开销大（stack trace）
    public void avoidExceptionOverhead() {
        // 推荐：返回错误码或Optional
        Optional<String> result = findUser(id);
        
        // 避免：频繁抛出异常
        // throw new BusinessException("not found");
    }
    
    // 5. 使用 Lambda 和 Stream 的性能注意
    // 简单场景：for 循环更快
    // 复杂操作：Stream 更可读，可接受的开销
    
    // 6. 并行流
    public long parallelSum(List<Long> numbers) {
        return numbers.parallelStream()
            .mapToLong(Long::longValue)
            .sum();
        // 注意：并行流适合大数据量，小数据量可能更慢
    }
    
    // 7. 使用缓存
    private final Cache<String, Data> cache = CacheBuilder.newBuilder()
        .maximumSize(10000)
        .expireAfterWrite(1, TimeUnit.HOURS)
        .build();
}
```

### 8.3.1 方法耗时监控

```java
public class PerformanceMonitor {
    // 使用代理/AOP记录方法耗时
    public class TimingAspect {
        public Object monitor(ProceedingJoinPoint joinPoint) throws Throwable {
            long start = System.nanoTime();
            try {
                return joinPoint.proceed();
            } finally {
                long duration = System.nanoTime() - start;
                if (duration > 1_000_000) { // 超过1ms
                    log.warn("方法 {} 耗时: {}ms", 
                        joinPoint.getSignature(), duration / 1_000_000);
                }
            }
        }
    }
    
    // 使用 Micrometer 指标监控
    public void registerMetrics(MeterRegistry registry) {
        Counter counter = Counter.builder("api.requests")
            .tag("endpoint", "/api/users")
            .register(registry);
        
        Timer timer = Timer.builder("api.latency")
            .publishPercentiles(0.5, 0.95, 0.99)
            .register(registry);
    }
}
```

### 8.3.2 内存泄漏排查

```java
public class MemoryLeakDetection {
    // 1. 常见泄漏场景
    // - 静态集合未清理
    // - 缓存无过期策略
    // - 监听器未注销
    // - 线程未关闭
    // - 连接未释放
    
    // 2. 使用 WeakReference 防止泄漏
    public class CacheWithWeakRef {
        private final Map<String, WeakReference<Data>> cache = new ConcurrentHashMap<>();
        
        public void put(String key, Data data) {
            cache.put(key, new WeakReference<>(data));
        }
        
        public Data get(String key) {
            WeakReference<Data> ref = cache.get(key);
            return ref != null ? ref.get() : null;
        }
    }
    
    // 3. Arthas 排查命令
    // watch com.example.Service * '{params,returnObj,throwExp}'
    // trace com.example.Service process
    // monitor com.example.Service *
    // heapdump /tmp/dump.hprof
    
    // 4. MAT 分析堆转储
    // - Histogram：对象统计
    // - Dominator Tree：支配树
    // - Path to GC Roots：引用链
}
```

## 8.4 JIT 编译与调优

```java
public class JITOptimization {
    // JIT (Just-In-Time) 编译：热点代码编译为本地机器码
    // - 方法调用次数超过阈值触发编译
    // - HotSpot 使用编译等级：0(解释执行) → 1(C1) → 2(C2)
    
    // 常用JIT参数
    // -XX:CompileThreshold=10000     编译触发阈值
    // -XX:+PrintCompilation          打印编译日志
    // -XX:+UnlockDiagnosticVMOptions
    // -XX:+PrintInlining             打印内联信息
    
    // 逃逸分析
    // -XX:+DoEscapeAnalysis          开启逃逸分析（默认开启）
    // 栈上分配：对象未逃逸出方法，分配在栈上
    // 标量替换：将对象拆解为基本类型
    // 锁消除：对象未逃逸出线程，消除锁
    
    public void stackAllocationDemo() {
        // Point 对象未逃逸，可能在栈上分配
        Point p = new Point(0, 0);
        p.setX(10);
        p.setY(20);
        System.out.println(p.getX() + p.getY());
        // 方法结束后，Point 自动回收，无需GC
    }
    
    // 循环优化
    public void loopOptimization() {
        // 循环展开：JIT 自动展开循环体
        // 循环不变代码外提
        int[] data = new int[1000000];
        int limit = data.length; // 外提
        int sum = 0;
        for (int i = 0; i < limit; i++) {
            sum += data[i];
        }
    }
    
    // 方法内联
    public int calculate(int a, int b) {
        return add(a, b); // JIT 可能内联 add 方法
    }
    
    // 小方法容易被内联
    private int add(int a, int b) {
        return a + b;
    }
}
```
